func (c *Client) processResponse(response *http.Response) ([]*entities.Coin, error) {
	var respMap map[string]map[string]float64

	if err := json.NewDecoder(response.Body).Decode(&respMap); err != nil {
		err := errors.Wrap(entities.ErrInternal, "json decode")
		slog.Error("Decode", "err", err)
		return nil, err
	}

	coins := make([]*entities.Coin, 0)
	if len(respMap) == 0 {
		err := errors.Wrap(entities.ErrNotFound, "result not contained data")
		slog.Error("processing result", "err", err)
		return nil, err
	}
	for title, costMap := range respMap {
		coin, err := entities.NewCoin(title, costMap[c.costIn])
		if err != nil {
			err := errors.Wrap(err, "new coin from response")
			slog.Error("NewCoin", "err", err)
			return nil, err
		}
		coins = append(coins, coin)
	}

	return coins, nil
}

func (c *Client) processStatusCode(response *http.Response) error {
	var respErr error
	switch response.StatusCode {
	case http.StatusBadRequest:
		respErr = entities.ErrInvalidParam
	case http.StatusNotFound:
		respErr = entities.ErrNotFound
	default:
		respErr = entities.ErrInternal
	}

	return errors.Wrap(respErr, "response status code")
}
